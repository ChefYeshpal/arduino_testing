const int trigpin = 2;
const int echopin = 3;
int buzzer = A2;

void setup() {
  Serial.begin(9600);
  pinMode(trigpin, OUTPUT); // trig is input
  pinMode(echopin, INPUT); // echo is output
  pinMode(buzzer, OUTPUT);
}

void loop () {
  digitalWrite(trigpin, LOW); //clear the trigpin
  delayMicroseconds(2);
  digitalWrite(trigpin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigpin, LOW);

  long duration = pulseIn(echopin, HIGH); // read the echo time
  int distance = duration * 0.034 / 2; // calculate distance in cm

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");
  delay(500);

  if (distance <= 30) {
    digitalWrite(buzzer, HIGH);
  } else {
    digitalWrite(buzzer, LOW);
  }

  delay(1000);
}
